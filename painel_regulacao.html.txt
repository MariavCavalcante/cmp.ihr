    <script>
        let currentPage = 1;
        const totalPages = 20;

        function updatePageDisplay() {
            const pages = document.querySelectorAll('.page');
            pages.forEach((p, index) => {
                p.classList.toggle('active', index === currentPage - 1);
            });

            document.getElementById('current-page-num').innerText = currentPage;
            document.getElementById('total-pages-num').innerText = totalPages;

            const pct = (currentPage / totalPages) * 100;
            document.getElementById('progress-bar').style.width = pct + '%';

            document.getElementById('btn-prev').disabled = (currentPage === 1);
            document.getElementById('btn-prev').classList.toggle('opacity-50', currentPage === 1);
            document.getElementById('btn-next').disabled = (currentPage === totalPages);
            document.getElementById('btn-next').classList.toggle('opacity-50', currentPage === totalPages);
        }

        function nextPage() {
            if (currentPage < totalPages) {
                currentPage++;
                updatePageDisplay();
            }
        }

        function prevPage() {
            if (currentPage > 1) {
                currentPage--;
                updatePageDisplay();
            }
        }

        function goToPage(pageNum) {
            currentPage = pageNum;
            updatePageDisplay();
        }

        function toggleFullScreen() {
            if (!document.fullscreenElement) {
                document.documentElement.requestFullscreen();
            } else {
                if (document.exitFullscreen) {
                    document.exitFullscreen();
                }
            }
        }

        document.addEventListener('keydown', (e) => {
            if (e.key === 'ArrowRight') nextPage();
            if (e.key === 'ArrowLeft') prevPage();
            if (e.key === 'Home') goToPage(1);
            if (e.key === 'End') goToPage(totalPages);
        });

        window.onload = function() {
            updatePageDisplay();
        };
    </script>
